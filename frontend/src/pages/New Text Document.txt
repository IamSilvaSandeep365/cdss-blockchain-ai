import { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import api from '../api/axios';
import { useAuth } from '../context/AuthContext';

export default function Dashboard() {
  const { user, logout } = useAuth();
  const navigate = useNavigate();

  const [patients, setPatients] = useState(0);
  const [stats, setStats]       = useState(null);
  const [health, setHealth]     = useState(null);
  const [loading, setLoading]   = useState(true);

  useEffect(() => {
    loadAll();
  }, []);

  const loadAll = async () => {
    setLoading(true);
    try {
      const [patientsRes, statsRes, healthRes] = await Promise.all([
        api.get('/patients'),
        api.get('/prescriptions/stats'),
        api.get('/health'),
      ]);
      setPatients(patientsRes.data.length);
      setStats(statsRes.data);
      setHealth(healthRes.data);
    } catch (err) {
      // If stats endpoint isn't ready, fail gracefully
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  const StatCard = ({ label, value, color }) => (
    <div className="bg-white rounded-2xl shadow p-6">
      <div className="text-sm text-slate-500">{label}</div>
      <div className={`text-3xl font-bold mt-1 ${color}`}>{value}</div>
    </div>
  );

  const HealthBadge = ({ label, up }) => (
    <div className="flex items-center justify-between py-2">
      <span className="text-slate-700">{label}</span>
      <span
        className={`text-xs font-medium px-2 py-1 rounded-full ${
          up ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'
        }`}
      >
        {up ? '● UP' : '● DOWN'}
      </span>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-100">
      <header className="bg-white shadow-sm">
        <div className="max-w-6xl mx-auto px-6 py-4 flex justify-between items-center">
          <div>
            <h1 className="text-xl font-bold text-slate-800">CDSS Dashboard</h1>
            <p className="text-xs text-slate-500">
              {user?.fullName || user?.username} · {user?.role}
            </p>
          </div>
          <div className="flex gap-2">
            <button
              onClick={() => navigate('/patients')}
              className="rounded-lg bg-blue-600 px-4 py-2 text-sm text-white hover:bg-blue-700 transition"
            >
              Patients
            </button>
            <button
              onClick={() => { logout(); navigate('/login'); }}
              className="rounded-lg bg-slate-200 px-4 py-2 text-sm text-slate-700 hover:bg-slate-300 transition"
            >
              Logout
            </button>
          </div>
        </div>
      </header>

      <main className="max-w-6xl mx-auto px-6 py-8">
        {loading ? (
          <div className="text-center text-slate-500 py-8">Loading dashboard…</div>
        ) : (
          <>
            {/* Stat cards */}
            <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-6">
              <StatCard label="Total Patients" value={patients} color="text-blue-600" />
              <StatCard label="Prescriptions" value={stats?.totalPrescriptions ?? 0} color="text-slate-800" />
              <StatCard label="On Blockchain" value={stats?.onBlockchain ?? 0} color="text-green-600" />
              <StatCard label="Pending" value={stats?.pending ?? 0} color="text-amber-600" />
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
              {/* System health */}
              <div className="bg-white rounded-2xl shadow p-6">
                <h2 className="text-lg font-semibold text-slate-800 mb-4">
                  System Health
                </h2>
                <HealthBadge label="Spring Boot Backend" up={health?.springBoot === 'UP'} />
                <HealthBadge label="Flask AI Service" up={health?.flaskAI === 'UP'} />
                <HealthBadge label="Blockchain (Ethereum)" up={health?.blockchain === 'UP'} />
                <div className="mt-3 pt-3 border-t border-slate-100 text-sm text-slate-500">
                  Blockchain records: {health?.totalBlockchainRecords ?? 0}
                </div>
              </div>

              {/* Disease breakdown */}
              <div className="bg-white rounded-2xl shadow p-6">
                <h2 className="text-lg font-semibold text-slate-800 mb-4">
                  Diagnoses Breakdown
                </h2>
                {stats?.byDisease && Object.keys(stats.byDisease).length > 0 ? (
                  <div className="space-y-2">
                    {Object.entries(stats.byDisease).map(([disease, count]) => (
                      <div key={disease} className="flex justify-between items-center text-sm">
                        <span className="text-slate-700">{disease}</span>
                        <span className="bg-slate-100 text-slate-700 px-2 py-0.5 rounded-full text-xs font-medium">
                          {count}
                        </span>
                      </div>
                    ))}
                  </div>
                ) : (
                  <p className="text-sm text-slate-500">No diagnoses yet.</p>
                )}
              </div>
            </div>
          </>
        )}
      </main>
    </div>
  );
}
